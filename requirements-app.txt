# Demo application (app/). Install on top of requirements.txt:

# Streamlit is pinned: the app uses st.navigation, segmented_control and
# width="stretch", which older releases lack. These exact versions are the
# ones the app was tested with.
streamlit==1.67.0
plotly==7.1.0

# pyarrow is pinned because Streamlit otherwise pulls the newest release,
# which requires NumPy >= 2: on a NumPy 1.x venv every table in the app
# crashes. 21.0.0 supports both NumPy 1.x and 2.x (tested with 1.26 and 2.5).
pyarrow==21.0.0
networkx>=3.0
scipy>=1.10
